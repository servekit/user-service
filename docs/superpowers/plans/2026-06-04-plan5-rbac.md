```go
// Authorizer checks user permissions.
type Authorizer interface {
	// Check returns true if the user has the specified resource:action permission.
	Check(ctx context.Context, userID int64, resource, action string) (bool, error)

	// CheckInGroup checks permission within a specific group scope.
	CheckInGroup(ctx context.Context, userID, groupID int64, resource, action string) (bool, error)

	// Refresh clears and reloads the permission cache.
	Refresh(ctx context.Context) error
}
```

这是权限系统的核心查询，从设计文档 §12.3：

```go
package rbac

import (
	"context"
	"fmt"

	"gorm.io/gorm"
)

// PermissionEntry represents a resource:action pair (used for SQL scan results).
type PermissionEntry struct {
	Resource string
	Action   string
}

// Checker queries user permissions from PostgreSQL.
type Checker struct {
	db *gorm.DB
}

func NewChecker(db *gorm.DB) *Checker {
	return &Checker{db: db}
}

// GetUserPermissions returns all permissions for a user (direct + group-inherited).
func (c *Checker) GetUserPermissions(ctx context.Context, userID int64) ([]PermissionEntry, error) {
	query := `
		WITH all_roles AS (
			SELECT role_id FROM user_roles WHERE user_id = ?
			UNION
			SELECT gr.role_id
			FROM user_groups ug
			JOIN group_roles gr ON gr.group_id = ug.group_id
			WHERE ug.user_id = ?
		)
		SELECT DISTINCT p.resource, p.action
		FROM all_roles ar
		JOIN (
			SELECT rp.role_id, p.resource, p.action
			FROM role_permissions rp
			JOIN permissions p ON p.id = rp.permission_id
			UNION
			SELECT rpg.role_id, p.resource, p.action
			FROM role_permission_groups rpg
			JOIN permission_group_items pgi ON pgi.permission_group_id = rpg.permission_group_id
			JOIN permissions p ON p.id = pgi.permission_id
		) p ON p.role_id = ar.role_id
	`
	var perms []PermissionEntry
	if err := c.db.WithContext(ctx).Raw(query, userID, userID).Scan(&perms).Error; err != nil {
		return nil, fmt.Errorf("query user permissions: %w", err)
	}
	return perms, nil
}

// GetUserRoles returns all role IDs for a user (direct + group-inherited).
func (c *Checker) GetUserRoles(ctx context.Context, userID int64) ([]int64, error) {
	query := `
		SELECT role_id FROM user_roles WHERE user_id = ?
		UNION
		SELECT gr.role_id
		FROM user_groups ug
		JOIN group_roles gr ON gr.group_id = ug.group_id
		WHERE ug.user_id = ?
	`
	var roleIDs []int64
	if err := c.db.WithContext(ctx).Raw(query, userID, userID).Scan(&roleIDs).Error; err != nil {
		return nil, fmt.Errorf("query user roles: %w", err)
	}
	return roleIDs, nil
}

// GetGroupRoles returns all role IDs assigned to a group.
func (c *Checker) GetGroupRoles(ctx context.Context, groupID int64) ([]int64, error) {
	var roleIDs []int64
	if err := c.db.WithContext(ctx).Model(&struct{}{}).
		Table("group_roles").
		Where("group_id = ?", groupID).
		Pluck("role_id", &roleIDs).Error; err != nil {
		return nil, fmt.Errorf("query group roles: %w", err)
	}
	return roleIDs, nil
}

// GetUserIDsByRole returns all user IDs that have a specific role.
func (c *Checker) GetUserIDsByRole(ctx context.Context, roleID int64) ([]int64, error) {
	query := `
		SELECT user_id FROM user_roles WHERE role_id = ?
		UNION
		SELECT ug.user_id
		FROM user_groups ug
		JOIN group_roles gr ON gr.group_id = ug.group_id
		WHERE gr.role_id = ?
	`
	var userIDs []int64
	if err := c.db.WithContext(ctx).Raw(query, roleID, roleID).Scan(&userIDs).Error; err != nil {
		return nil, fmt.Errorf("query users by role: %w", err)
	}
	return userIDs, nil
}
```

```go
package rbac

import (
	"context"
	"fmt"
	"time"

	"user-service/internal/config"

	"github.com/redis/go-redis/v9"
)

// Cache manages RBAC data in Redis.
type Cache struct {
	client *redis.Client
	cfg    config.RBACConfig
}

func NewCache(client *redis.Client, cfg config.RBACConfig) *Cache {
	return &Cache{client: client, cfg: cfg}
}

// UserCacheKey returns the Redis key for user permissions.
func (c *Cache) userPermsKey(userID int64) string {
	return fmt.Sprintf("%s%d", c.cfg.Redis.UserPermsPrefix, userID)
}

func (c *Cache) userRolesKey(userID int64) string {
	return fmt.Sprintf("%s%d", c.cfg.Redis.UserRolesPrefix, userID)
}

func (c *Cache) rolePermsKey(roleID int64) string {
	return fmt.Sprintf("%s%d", c.cfg.Redis.RolePermsPrefix, roleID)
}

func (c *Cache) groupRolesKey(groupID int64) string {
	return fmt.Sprintf("%s%d", c.cfg.Redis.GroupRolesPrefix, groupID)
}

// GetUserPermissions checks Redis cache first, returns nil if miss.
func (c *Cache) GetUserPermissions(ctx context.Context, userID int64) (map[string]bool, error) {
	key := c.userPermsKey(userID)
	result, err := c.client.HGetAll(ctx, key).Result()
	if err != nil {
		return nil, fmt.Errorf("redis hgetall: %w", err)
	}
	if len(result) == 0 {
		return nil, nil // cache miss
	}
	perms := make(map[string]bool, len(result))
	for k := range result {
		perms[k] = true
	}
	return perms, nil
}

// SetUserPermissions writes user permissions to Redis with TTL.
func (c *Cache) SetUserPermissions(ctx context.Context, userID int64, perms []PermissionEntry) error {
	key := c.userPermsKey(userID)
	if len(perms) == 0 {
		// Store an empty marker to distinguish from cache miss
		c.client.HSet(ctx, key, "__empty", "1")
	} else {
		fields := make(map[string]any, len(perms))
		for _, p := range perms {
			fields[p.Resource+":"+p.Action] = "1"
		}
		c.client.HSet(ctx, key, fields)
	}
	c.client.Expire(ctx, key, c.cfg.Cache.UserPermsTTL)
	return nil
}

// InvalidateUser removes user permission and role caches.
func (c *Cache) InvalidateUser(ctx context.Context, userID int64) error {
	c.client.Del(ctx, c.userPermsKey(userID))
	c.client.Del(ctx, c.userRolesKey(userID))
	return nil
}

// InvalidateRole removes role permission cache + all affected user caches.
func (c *Cache) InvalidateRole(ctx context.Context, roleID int64, affectedUserIDs []int64) error {
	c.client.Del(ctx, c.rolePermsKey(roleID))
	for _, uid := range affectedUserIDs {
		c.InvalidateUser(ctx, uid)
	}
	return nil
}

// InvalidateGroup removes group role cache + all affected member caches.
func (c *Cache) InvalidateGroup(ctx context.Context, groupID int64, memberIDs []int64) error {
	c.client.Del(ctx, c.groupRolesKey(groupID))
	for _, uid := range memberIDs {
		c.InvalidateUser(ctx, uid)
	}
	return nil
}
```

```go
// DBAuthorizer implements Authorizer using PostgreSQL + Redis cache.
type DBAuthorizer struct {
	checker *Checker
	cache   *Cache
}

func NewDBAuthorizer(checker *Checker, cache *Cache) *DBAuthorizer {
	return &DBAuthorizer{checker: checker, cache: cache}
}

func (a *DBAuthorizer) Check(ctx context.Context, userID int64, resource, action string) (bool, error) {
	permKey := resource + ":" + action

	// Check cache first
	cached, err := a.cache.GetUserPermissions(ctx, userID)
	if err != nil {
		// Log but don't fail — fall through to DB
		cached = nil
	}
	if cached != nil {
		return cached[permKey], nil
	}

	// Cache miss — query DB
	perms, err := a.checker.GetUserPermissions(ctx, userID)
	if err != nil {
		return false, fmt.Errorf("query permissions: %w", err)
	}

	// Write to cache
	a.cache.SetUserPermissions(ctx, userID, perms)

	// Check the result
	for _, p := range perms {
		if p.Resource == resource && p.Action == action {
			return true, nil
		}
	}
	return false, nil
}

func (a *DBAuthorizer) CheckInGroup(ctx context.Context, userID, groupID int64, resource, action string) (bool, error) {
	// For first version, CheckInGroup uses the same permission set
	// (group scope filtering is a future enhancement)
	return a.Check(ctx, userID, resource, action)
}

func (a *DBAuthorizer) Refresh(ctx context.Context) error {
	// Refresh is a no-op — individual invalidation happens on mutations
	// via cache.InvalidateUser/Role/Group
	return nil
}
```

```go
func Seed(ctx context.Context, db *gorm.DB) error {
	// 1. Built-in permissions
	perms := []struct {
		Resource    string
		Action      string
		Description string
	}{
		{"users", "read", "查看用户列表/详情"},
		{"users", "write", "禁用/启用用户"},
		{"users", "delete", "删除用户"},
		{"groups", "read", "查看群组"},
		{"groups", "write", "创建/编辑群组"},
		{"groups", "delete", "删除群组"},
		{"groups", "manage", "管理群组成员"},
		{"roles", "read", "查看角色"},
		{"roles", "write", "创建/编辑角色"},
		{"roles", "delete", "删除角色"},
		{"permissions", "read", "查看权限列表"},
		{"login_logs", "read", "查看登录日志"},
		{"sessions", "manage", "管理会话"},
		{"user_roles", "write", "分配用户角色"},
		{"user_roles", "delete", "撤销用户角色"},
	}
	for _, p := range perms {
		db.WithContext(ctx).Clauses(clause.OnConflict{DoNothing: true}).
			Table("permissions").
			Create(map[string]any{
				"resource": p.Resource, "action": p.Action,
				"description": p.Description, "is_builtin": true,
			})
	}

	// 2. Built-in permission groups
	groups := []struct {
		Name        string
		Description string
		Perms       []struct{ Resource, Action string }
	}{
		{"user_management", "用户管理", []struct{ Resource, Action string }{
			{"users", "read"}, {"users", "write"}, {"users", "delete"},
		}},
		{"group_management", "群组管理", []struct{ Resource, Action string }{
			{"groups", "read"}, {"groups", "write"}, {"groups", "delete"}, {"groups", "manage"},
		}},
		{"role_management", "角色管理", []struct{ Resource, Action string }{
			{"roles", "read"}, {"roles", "write"}, {"roles", "delete"}, {"permissions", "read"},
		}},
		{"security", "安全管理", []struct{ Resource, Action string }{
			{"login_logs", "read"}, {"sessions", "manage"},
		}},
	}
	for _, g := range groups {
		db.WithContext(ctx).Clauses(clause.OnConflict{DoNothing: true}).
			Table("permission_groups").
			Create(map[string]any{"name": g.Name, "description": g.Description, "is_builtin": true})
		// Look up group ID and permission IDs, create permission_group_items
		// (use ON CONFLICT DO NOTHING for idempotency)
	}

	// 3. Built-in roles
	roles := []struct {
		Name        string
		Description string
		PermKeys    []string // resource:action pairs for direct binding
		GroupNames  []string // permission group names
	}{
		{
			"super_admin", "超级管理员",
			nil, // gets all permissions via groups
			[]string{"user_management", "group_management", "role_management", "security"},
		},
		{
			"admin", "管理员",
			nil,
			[]string{"user_management", "group_management", "security"},
		},
		{
			"member", "普通成员",
			nil, nil, // no permissions
		},
		{
			"group_admin", "群组管理员",
			[]string{"groups:read", "groups:manage"},
			nil,
		},

			Table("roles").
			Create(map[string]any{"name": r.Name, "description": r.Description, "is_builtin": true})

	}
```

```go
import (
	"context"
	"fmt"
	"strings"

	"user-service/internal/rbac"
	"user-service/internal/session"

	"google.golang.org/grpc"
	"google.golang.org/grpc/codes"
	"google.golang.org/grpc/metadata"
	"google.golang.org/grpc/status"
)

type contextKey string

const (
	UserIDKey contextKey = "user_id"
)

// Admin permission mapping from design doc §12.5.
var adminPermissions = map[string]struct{ resource, action string }{
	"/user.UserService/GetUser":         {"users", "read"},
	"/user.UserService/ListUsers":       {"users", "read"},
	"/user.UserService/DisableUser":     {"users", "write"},
	"/user.UserService/GetLoginLogs":    {"login_logs", "read"},
	"/user.UserService/CreateGroup":     {"groups", "write"},
	"/user.UserService/GetGroup":        {"groups", "read"},
	"/user.UserService/ListGroups":      {"groups", "read"},
	"/user.UserService/UpdateGroup":     {"groups", "write"},
	"/user.UserService/DeleteGroup":     {"groups", "delete"},
	"/user.UserService/AddGroupMember":  {"groups", "manage"},
	"/user.UserService/AddGroupRole":    {"groups", "manage"},
	"/user.UserService/RemoveGroupRole": {"groups", "manage"},
	"/user.UserService/ListGroupRoles":  {"groups", "read"},
	"/user.UserService/CreateRole":      {"roles", "write"},
	"/user.UserService/UpdateRole":      {"roles", "write"},
	"/user.UserService/DeleteRole":      {"roles", "delete"},
	"/user.UserService/ListRoles":       {"roles", "read"},
	"/user.UserService/AssignRole":           {"user_roles", "write"},
	"/user.UserService/RevokeRole":           {"user_roles", "delete"},
	"/user.UserService/ListUserRoles":        {"user_roles", "read"},
	"/user.UserService/ListPermissions":      {"permissions", "read"},
	"/user.UserService/ListPermissionGroups": {"permissions", "read"},
	"/user.UserService/ListGroupMembers":     {"groups", "read"},
	"/user.UserService/RemoveGroupMember":    {"groups", "manage"},
}

// AuthInterceptor returns a gRPC unary interceptor that validates sessions
// and checks RBAC permissions for admin APIs.
func AuthInterceptor(sessionMgr *session.Manager, authorizer rbac.Authorizer) grpc.UnaryServerInterceptor {
	return func(ctx context.Context, req any, info *grpc.UnaryServerInfo, handler grpc.UnaryHandler) (any, error) {
		// Extract session ID from Authorization header
		sessionID, err := extractBearerToken(ctx)
		if err != nil {
			return nil, status.Error(codes.Unauthenticated, err.Error())
		}

		// Validate session
		data, err := sessionMgr.Get(ctx, sessionID)
		if err != nil {
			return nil, status.Error(codes.Unauthenticated, "invalid or expired session")
		}

		// Inject user ID into context
		ctx = context.WithValue(ctx, UserIDKey, data.UserID)

		// Check RBAC for admin APIs
		if perm, ok := adminPermissions[info.FullMethod]; ok {
			allowed, err := authorizer.Check(ctx, data.UserID, perm.resource, perm.action)
			if err != nil {
				return nil, status.Error(codes.Internal, "permission check failed")
			}
			if !allowed {
				return nil, status.Error(codes.PermissionDenied, "permission denied")
			}
		}

		return handler(ctx, req)
	}
}

// GetUserID extracts the user ID from context (set by interceptor).
func GetUserID(ctx context.Context) (int64, error) {
	userID, ok := ctx.Value(UserIDKey).(int64)
	if !ok {
		return 0, fmt.Errorf("user ID not found in context")
	}
	return userID, nil
}

func extractBearerToken(ctx context.Context) (string, error) {
	md, ok := metadata.FromIncomingContext(ctx)
	if !ok {
		return "", fmt.Errorf("no metadata found")
	}
	values := md.Get("authorization")
	if len(values) == 0 {
		return "", fmt.Errorf("authorization header missing")
	}
	parts := strings.SplitN(values[0], " ", 2)
	if len(parts) != 2 || !strings.EqualFold(parts[0], "bearer") {
		return "", fmt.Errorf("invalid authorization header format")
	}
	return parts[1], nil
}
```
